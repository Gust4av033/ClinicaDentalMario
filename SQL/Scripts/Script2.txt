-- Vista: Pacientes Activos
CREATE OR ALTER VIEW Pacientes.vwPacientes AS
SELECT IdPaciente, NombreCompleto, Direccion, FechaNacimiento, Sexo, DUI, Telefono, 
       NombreEncargado, ContactoEmergencia, TelefonoEmergencia, FechaRegistro
FROM Pacientes.Pacientes
WHERE Activo = 1;
GO

-- Vista: Doctores Activos
CREATE OR ALTER VIEW Personal.vwDoctores AS
SELECT IdDoctor, NombreCompleto, Especialidad, Telefono, Correo, Direccion, NumeroJVPO
FROM Personal.Doctores
WHERE Activo = 1;
GO

-- Vista: Agenda de Hoy
CREATE OR ALTER VIEW Agenda.vwAgendaHoy AS
SELECT c.IdCita, p.NombreCompleto AS Paciente, d.NombreCompleto AS Doctor, 
       c.FechaHora, e.Nombre AS Estado, c.Observaciones
FROM Agenda.Citas c
INNER JOIN Pacientes.Pacientes p ON c.IdPaciente = p.IdPaciente
INNER JOIN Personal.Doctores d ON c.IdDoctor = d.IdDoctor
INNER JOIN Catalogos.EstadosCita e ON c.IdEstado = e.IdEstado
WHERE CAST(c.FechaHora AS DATE) = CAST(GETDATE() AS DATE);
GO

-- Vista: Agenda de la Semana
CREATE OR ALTER VIEW Agenda.vwAgendaSemana AS
SELECT c.IdCita, p.NombreCompleto AS Paciente, d.NombreCompleto AS Doctor, 
       c.FechaHora, e.Nombre AS Estado, c.Observaciones
FROM Agenda.Citas c
INNER JOIN Pacientes.Pacientes p ON c.IdPaciente = p.IdPaciente
INNER JOIN Personal.Doctores d ON c.IdDoctor = d.IdDoctor
INNER JOIN Catalogos.EstadosCita e ON c.IdEstado = e.IdEstado
WHERE c.FechaHora >= CAST(GETDATE() AS DATE) 
  AND c.FechaHora <= DATEADD(day, 7, CAST(GETDATE() AS DATE));
GO

-- Vista: Catálogo de Tratamientos Activos
CREATE OR ALTER VIEW Catalogos.vwTratamientos AS
SELECT IdTratamiento, Nombre, Descripcion, PrecioBase, DuracionMinutos
FROM Catalogos.CatalogoTratamientos
WHERE Activo = 1;
GO

-- Vista: Tratamientos Pendientes o en Progreso
CREATE OR ALTER VIEW Odontologia.vwTratamientosPendientes AS
SELECT tp.Id AS IdTratamientoPaciente, p.NombreCompleto AS Paciente, 
       ct.Nombre AS Tratamiento, tp.CostoTotal, tp.Estado, tp.FechaInicio
FROM Odontologia.TratamientosPaciente tp
INNER JOIN Pacientes.Pacientes p ON tp.IdPaciente = p.IdPaciente
INNER JOIN Catalogos.CatalogoTratamientos ct ON tp.IdTratamiento = ct.IdTratamiento
WHERE tp.Estado IN ('Pendiente', 'En progreso');
GO

-- Vista: Todos los Pagos
CREATE OR ALTER VIEW Odontologia.vwPagos AS
SELECT pg.IdPago, p.NombreCompleto AS Paciente, ct.Nombre AS Tratamiento, 
       pg.FechaPago, pg.Monto, pg.MetodoPago, pg.Observacion
FROM Odontologia.Pagos pg
INNER JOIN Odontologia.TratamientosPaciente tp ON pg.IdTratamientoPaciente = tp.Id
INNER JOIN Pacientes.Pacientes p ON tp.IdPaciente = p.IdPaciente
INNER JOIN Catalogos.CatalogoTratamientos ct ON tp.IdTratamiento = ct.IdTratamiento;
GO

-- Vista: Ingresos Diarios
CREATE OR ALTER VIEW Facturacion.vwIngresosDiarios AS
SELECT CAST(FechaPago AS DATE) AS Fecha, SUM(Monto) AS TotalIngresos
FROM Odontologia.Pagos
GROUP BY CAST(FechaPago AS DATE);
GO

-- Vista: Ingresos Mensuales
CREATE OR ALTER VIEW Facturacion.vwIngresosMensuales AS
SELECT YEAR(FechaPago) AS Anio, MONTH(FechaPago) AS Mes, SUM(Monto) AS TotalIngresos
FROM Odontologia.Pagos
GROUP BY YEAR(FechaPago), MONTH(FechaPago);
GO

-- Vista: Historial Clínico Integral
CREATE OR ALTER VIEW Pacientes.vwHistorialPaciente AS
SELECT hc.IdHistorial, p.NombreCompleto AS Paciente, d.NombreCompleto AS Doctor,
       hc.MotivoConsulta, hc.AntecedentesMedicos, hc.AntecedentesOdontologicos,
       hc.Diagnostico, hc.PlanTratamiento, hc.FechaConsulta
FROM Pacientes.HistorialClinico hc
INNER JOIN Pacientes.Pacientes p ON hc.IdPaciente = p.IdPaciente
INNER JOIN Personal.Doctores d ON hc.IdDoctor = d.IdDoctor;
GO

-- Vista: Saldo Total de Pacientes (Costo Total - Pagado)
CREATE OR ALTER VIEW Pacientes.vwSaldoPacientes AS
SELECT p.IdPaciente, p.NombreCompleto, 
       ISNULL(SUM(tp.CostoTotal), 0) AS TotalCargos,
       ISNULL((SELECT SUM(Monto) FROM Odontologia.Pagos pg 
               INNER JOIN Odontologia.TratamientosPaciente t2 ON pg.IdTratamientoPaciente = t2.Id 
               WHERE t2.IdPaciente = p.IdPaciente), 0) AS TotalPagado
FROM Pacientes.Pacientes p
LEFT JOIN Odontologia.TratamientosPaciente tp ON p.IdPaciente = tp.IdPaciente
GROUP BY p.IdPaciente, p.NombreCompleto;
GO

-- Vista: Odontograma de Pacientes
CREATE OR ALTER VIEW Odontologia.vwOdontogramaPaciente AS
SELECT o.IdRegistro, p.NombreCompleto AS Paciente, o.NumeroPieza, 
       ed.Nombre AS EstadoDental, ed.ColorHex, o.Observaciones, o.FechaRegistro
FROM Odontologia.Odontograma o
INNER JOIN Pacientes.Pacientes p ON o.IdPaciente = p.IdPaciente
INNER JOIN Catalogos.CatalogoEstadosDentales ed ON o.IdEstadoDental = ed.IdEstadoDental;
GO

-- Función: Calcular Edad
CREATE OR ALTER FUNCTION Pacientes.fnCalcularEdad (@FechaNacimiento DATE)
RETURNS INT
AS
BEGIN
    IF @FechaNacimiento IS NULL RETURN NULL;
    DECLARE @Edad INT = DATEDIFF(YEAR, @FechaNacimiento, GETDATE());
    IF (MONTH(@FechaNacimiento) > MONTH(GETDATE()) OR 
       (MONTH(@FechaNacimiento) = MONTH(GETDATE()) AND DAY(@FechaNacimiento) > DAY(GETDATE())))
        SET @Edad = @Edad - 1;
    RETURN @Edad;
END;
GO

-- Función: Total Abonado por Tratamiento
CREATE OR ALTER FUNCTION Odontologia.fnTotalAbonado (@IdTratamientoPaciente INT)
RETURNS DECIMAL(10,2)
AS
BEGIN
    DECLARE @Total DECIMAL(10,2);
    SELECT @Total = ISNULL(SUM(Monto), 0) FROM Odontologia.Pagos WHERE IdTratamientoPaciente = @IdTratamientoPaciente;
    RETURN @Total;
END;
GO

-- Función: Saldo Pendiente por Tratamiento
CREATE OR ALTER FUNCTION Odontologia.fnSaldoPendiente (@IdTratamientoPaciente INT)
RETURNS DECIMAL(10,2)
AS
BEGIN
    DECLARE @Costo DECIMAL(10,2), @Abonado DECIMAL(10,2);
    SELECT @Costo = CostoTotal FROM Odontologia.TratamientosPaciente WHERE Id = @IdTratamientoPaciente;
    SET @Abonado = Odontologia.fnTotalAbonado(@IdTratamientoPaciente);
    RETURN ISNULL(@Costo, 0) - @Abonado;
END;
GO

-- Función: Obtener Nombre Completo
CREATE OR ALTER FUNCTION Pacientes.fnNombreCompletoPaciente (@IdPaciente INT)
RETURNS NVARCHAR(150)
AS
BEGIN
    DECLARE @Nombre NVARCHAR(150);
    SELECT @Nombre = NombreCompleto FROM Pacientes.Pacientes WHERE IdPaciente = @IdPaciente;
    RETURN @Nombre;
END;
GO

-- Función: Próxima Cita de un Paciente
CREATE OR ALTER FUNCTION Agenda.fnProximaCita (@IdPaciente INT)
RETURNS DATETIME
AS
BEGIN
    DECLARE @Proxima DATETIME;
    SELECT TOP 1 @Proxima = FechaHora FROM Agenda.Citas 
    WHERE IdPaciente = @IdPaciente AND FechaHora >= GETDATE() AND IdEstado = (SELECT IdEstado FROM Catalogos.EstadosCita WHERE Nombre = 'Confirmada' OR Nombre = 'Pendiente')
    ORDER BY FechaHora ASC;
    RETURN @Proxima;
END;
GO

-- Función: Cantidad de Tratamientos por Paciente
CREATE OR ALTER FUNCTION Odontologia.fnCantidadTratamientosPaciente (@IdPaciente INT)
RETURNS INT
AS
BEGIN
    DECLARE @Cantidad INT;
    SELECT @Cantidad = COUNT(*) FROM Odontologia.TratamientosPaciente WHERE IdPaciente = @IdPaciente;
    RETURN @Cantidad;
END;
GO

-- ----------------- SEGURIDAD -----------------
CREATE OR ALTER PROCEDURE Seguridad.sp_Login
    @Usuario VARCHAR(50),
    @PasswordHash VARCHAR(255)
AS
BEGIN
    SELECT IdUsuario, IdRol, NombreCompleto, Correo 
    FROM Seguridad.Usuarios 
    WHERE Usuario = @Usuario AND PasswordHash = @PasswordHash AND Activo = 1;
END;
GO

CREATE OR ALTER PROCEDURE Seguridad.sp_CambiarPassword
    @IdUsuario INT,
    @NuevoPasswordHash VARCHAR(255)
AS
BEGIN
    UPDATE Seguridad.Usuarios SET PasswordHash = @NuevoPasswordHash WHERE IdUsuario = @IdUsuario;
END;
GO

CREATE OR ALTER PROCEDURE Seguridad.sp_CrearUsuario
    @IdRol INT, @NombreCompleto NVARCHAR(150), @Usuario VARCHAR(50), @Correo VARCHAR(120), @PasswordHash VARCHAR(255)
AS
BEGIN
    INSERT INTO Seguridad.Usuarios (IdRol, NombreCompleto, Usuario, Correo, PasswordHash)
    VALUES (@IdRol, @NombreCompleto, @Usuario, @Correo, @PasswordHash);
END;
GO

-- ----------------- PACIENTES -----------------
CREATE OR ALTER PROCEDURE Pacientes.sp_InsertarPaciente
    @NombreCompleto NVARCHAR(150), @Direccion NVARCHAR(250), @FechaNacimiento DATE, @Sexo VARCHAR(20),
    @DUI VARCHAR(20), @Telefono VARCHAR(20), @NombreEncargado NVARCHAR(150), 
    @ContactoEmergencia NVARCHAR(150), @TelefonoEmergencia VARCHAR(20)
AS
BEGIN
    INSERT INTO Pacientes.Pacientes (NombreCompleto, Direccion, FechaNacimiento, Sexo, DUI, Telefono, NombreEncargado, ContactoEmergencia, TelefonoEmergencia)
    VALUES (@NombreCompleto, @Direccion, @FechaNacimiento, @Sexo, @DUI, @Telefono, @NombreEncargado, @ContactoEmergencia, @TelefonoEmergencia);
    SELECT SCOPE_IDENTITY() AS IdGenerado;
END;
GO

CREATE OR ALTER PROCEDURE Pacientes.sp_EditarPaciente
    @IdPaciente INT, @NombreCompleto NVARCHAR(150), @Direccion NVARCHAR(250), @FechaNacimiento DATE, 
    @Sexo VARCHAR(20), @DUI VARCHAR(20), @Telefono VARCHAR(20), @NombreEncargado NVARCHAR(150), 
    @ContactoEmergencia NVARCHAR(150), @TelefonoEmergencia VARCHAR(20)
AS
BEGIN
    UPDATE Pacientes.Pacientes
    SET NombreCompleto = @NombreCompleto, Direccion = @Direccion, FechaNacimiento = @FechaNacimiento, 
        Sexo = @Sexo, DUI = @DUI, Telefono = @Telefono, NombreEncargado = @NombreEncargado, 
        ContactoEmergencia = @ContactoEmergencia, TelefonoEmergencia = @TelefonoEmergencia
    WHERE IdPaciente = @IdPaciente;
END;
GO

CREATE OR ALTER PROCEDURE Pacientes.sp_EliminarPaciente
    @IdPaciente INT
AS
BEGIN
    -- Soft Delete (Solo desactiva)
    UPDATE Pacientes.Pacientes SET Activo = 0 WHERE IdPaciente = @IdPaciente;
END;
GO

CREATE OR ALTER PROCEDURE Pacientes.sp_BuscarPaciente
    @Termino NVARCHAR(150)
AS
BEGIN
    SELECT * FROM Pacientes.vwPacientes
    WHERE NombreCompleto LIKE '%' + @Termino + '%' OR DUI LIKE '%' + @Termino + '%';
END;
GO

CREATE OR ALTER PROCEDURE Pacientes.sp_ListarPacientes AS
BEGIN SELECT * FROM Pacientes.vwPacientes; END;
GO

BEGIN
