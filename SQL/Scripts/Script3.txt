-- 🛡️ 1. TABLA DE AUDITORÍA (BITÁCORA)
-- ==============================================================================
IF NOT EXISTS (SELECT * FROM sys.tables WHERE name = 'Bitacora' AND schema_id = SCHEMA_ID('Seguridad'))
BEGIN
    CREATE TABLE Seguridad.Bitacora (
        IdBitacora INT IDENTITY(1,1) PRIMARY KEY,
        Usuario VARCHAR(100) NOT NULL, -- Usuario de la Base de Datos o Aplicación
        Accion VARCHAR(50) NOT NULL,   -- INSERT, UPDATE, DELETE, LOGIN
        Tabla VARCHAR(100) NOT NULL,
        Fecha DATETIME DEFAULT GETDATE(),
        Equipo VARCHAR(100) NULL,      -- Nombre de la PC
        IP VARCHAR(50) NULL,           -- Dirección IP
        RegistroAfectado NVARCHAR(MAX) NULL -- Detalle de lo que se modificó
    );
END
GO

-- Trigger: Insertar Paciente
CREATE OR ALTER TRIGGER Pacientes.TR_InsertPaciente
ON Pacientes.Pacientes
AFTER INSERT
AS
BEGIN
    INSERT INTO Seguridad.Bitacora (Usuario, Accion, Tabla, Equipo, IP, RegistroAfectado)
    SELECT SYSTEM_USER, 'INSERT', 'Pacientes.Pacientes', HOST_NAME(), CAST(CONNECTIONPROPERTY('client_net_address') AS VARCHAR(50)),
           'IdPaciente: ' + CAST(i.IdPaciente AS VARCHAR(10)) + ' | Nombre: ' + i.NombreCompleto
    FROM inserted i;
END;
GO

-- (Opcional Integrado) Modificamos el SP de Login del Módulo 2 para auditar el TR_Login
CREATE OR ALTER PROCEDURE Seguridad.sp_Login
    @Usuario VARCHAR(50),
    @PasswordHash VARCHAR(255)
AS
BEGIN
    DECLARE @IdUsuario INT, @NombreCompleto NVARCHAR(150);
    
    SELECT @IdUsuario = IdUsuario, @NombreCompleto = NombreCompleto 
    FROM Seguridad.Usuarios 
    WHERE Usuario = @Usuario AND PasswordHash = @PasswordHash AND Activo = 1;

    IF @IdUsuario IS NOT NULL
    BEGIN
        -- Registro en Bitácora de acceso exitoso
        INSERT INTO Seguridad.Bitacora (Usuario, Accion, Tabla, Equipo, IP, RegistroAfectado)
        VALUES (@Usuario, 'LOGIN', 'Seguridad.Usuarios', HOST_NAME(), CAST(CONNECTIONPROPERTY('client_net_address') AS VARCHAR(50)), 'Acceso exitoso al sistema. ID: ' + CAST(@IdUsuario AS VARCHAR(10)));
        
        -- Retornamos los datos al sistema C#
        SELECT IdUsuario, IdRol, NombreCompleto, Correo 
        FROM Seguridad.Usuarios 
        WHERE IdUsuario = @IdUsuario;
    END
    ELSE
    BEGIN
        -- Registro en Bitácora de intento fallido
        INSERT INTO Seguridad.Bitacora (Usuario, Accion, Tabla, Equipo, IP, RegistroAfectado)
        VALUES (@Usuario, 'LOGIN FALLIDO', 'Seguridad.Usuarios', HOST_NAME(), CAST(CONNECTIONPROPERTY('client_net_address') AS VARCHAR(50)), 'Intento de acceso fallido.');
    END
END;
GO
